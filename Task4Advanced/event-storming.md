# Event Storming

```mermaid
flowchart LR
  classDef event fill:#f6a623,stroke:#333,color:#111;
  classDef subscriber fill:#d8ecff,stroke:#2b6cb0,color:#111;

  subgraph patient["Источник: Пациентский поток"]
    e1["Зарегистрирован новый пациент"]:::event
    e2["Создана запись на приём"]:::event
    e3["Завершён визит пациента"]:::event
  end

  subgraph finance["Источник: Финансовые сервисы"]
    e4["Создан кредитный договор"]:::event
    e5["Выставлен счёт"]:::event
    e6["Получен платёж"]:::event
  end

  subgraph ai["Источник: ИИ-исследования"]
    e7["Пройдено исследование ИИ"]:::event
  end

  subgraph partners["Источник: Новые направления"]
    e8["Зарегистрирован фарма-партнёр"]:::event
    e9["Получена телеметрия медоборудования"]:::event
  end

  subgraph platform["Подписчики"]
    s1["Данные и аналитика"]:::subscriber
    s2["Финансовые сервисы"]:::subscriber
    s3["Пациентский поток"]:::subscriber
    s4["Доступ и аудит"]:::subscriber
  end

  e1 --> s1
  e2 --> s1
  e3 --> s1
  e3 --> s2
  e4 --> s1
  e5 --> s1
  e6 --> s1
  e6 --> s3
  e7 --> s1
  e7 --> s3
  e8 --> s1
  e9 --> s1
  e9 --> s4
```
